//// =====================
//  FLUJO A - Sucursales, Areas, Mesas, Asignaciones
//// =====================
Table catalogos.Sucursal {
  id_sucursal int [pk, increment]
  codigo_sucursal nvarchar(20) [not null, unique]
  nombre nvarchar(100) [not null]
  telefono nvarchar(15)
  direccion nvarchar(100)
  es_activa bit [not null, default: `1`]
  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
  updated_at datetime2
}

Table catalogos.Area {
  id_area int [pk, not null, increment]
  sucursal_id int [not null]
  nombre nvarchar(50) [not null]
  descripcion nvarchar(100)
  es_activa bit [not null, default: `1`]
  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
}

Table catalogos.Mesa {
  id_mesa int [pk, increment]
  sucursal_id int [not null]
  area_id int [not null]
  codigo_mesa nvarchar(20) [not null]
  capacidad int [not null]
  es_activa bit [not null, default: `1`]
  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
  updated_at datetime2

  indexes {
    (sucursal_id, codigo_mesa) [unique, name: 'UQ_Mesa_Sucursal_Codigo']
    (sucursal_id, area_id)     [name: 'IX_Mesa_SucursalArea']
  }
}

Table operaciones.AsignacionMesa {
  id_asignacion_mesa bigint [pk, increment]
  mesa_id int         [not null]
  usuario_id bigint   [not null] // FK -> seguridad.Usuario (Mesero)
  es_activa bit       [not null, default: `1`]

  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
  updated_at datetime2

  indexes {
    (mesa_id, usuario_id, es_activa) [unique, name: 'UX_AsigMesa_MesaUsuarioActiva']
    (mesa_id, es_activa) [name: 'IX_AsigMesa_MesaActiva']
    (usuario_id, es_activa)           [name: 'IX_AsigMesa_MeseroActivo']
  }
  Note: 'Permite varios meseros activos por mesa'
}

Table operaciones.HoldMesa {
  id_hold_mesa bigint [pk, increment]
  //sucursal_id int     [not null] // FK -> catalogos.Sucursal
  mesa_id int         [not null] // FK -> catalogos.Mesa
  actor_tipo tinyint  [not null] // 1=Cliente, 2=Recepción
  actor_usuario_id bigint // opcional: FK -> seguridad.Usuario (quien originó)
  inicio datetime2    [not null] // ventana solicitada (inicio)
  fin_estimado datetime2 [not null] // ventana solicitada (fin estimado)
  expires_at datetime2 [not null] // TTL: cuando expira el hold
  estatus tinyint     [not null, default: `1`] // 1=Activo,2=Confirmado,3=Expirado,4=Cancelado
  notas nvarchar(300)
  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
  updated_at datetime2
  rowversion rowversion

  indexes {
    (mesa_id, expires_at) [name: 'IX_HoldMesa_Expira']
    (mesa_id, estatus)     [name: 'IX_HoldMesa_Estatus']
  }
}

Table operaciones.Reserva {
  id_reserva bigint [pk, increment]
  //sucursal_id int     [not null]          // FK -> catalogos.Sucursal
  //mesa_id int         [not null]          // FK -> catalogos.Mesa
  cliente_id bigint   // FK -> seguridad.Usuario (cliente)
  recepcionista_id bigint                  // FK -> seguridad.Usuario (si aplica)
  inicio datetime2    [not null]
  fin_estimado datetime2 [not null]
  estatus tinyint     [not null, default: `1`] // 1=Programada,2=EnCurso,3=Completada,4=NoShow,5=Cancelada
  tolerancia_min int                      // minutos para NoShow (nullable -> fallback config)
  notas nvarchar(300)
  hold_id bigint                          // FK opcional -> operaciones.HoldMesa (si se originó desde hold)
  created_at datetime2 [not null, default: `SYSUTCDATETIME()`]
  updated_at datetime2
